-- Une période : créations par département au 1er trimestre 2023

        FROM fait_annonce a
        WHERE a.type_annonce = 'creation' AND a.jour BETWEEN DATE '2023-01-01' AND DATE '2023-03-31'
        GROUP BY 1 ORDER BY 2 DESC
